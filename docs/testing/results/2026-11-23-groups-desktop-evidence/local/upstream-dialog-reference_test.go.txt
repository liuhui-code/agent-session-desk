package ui

import (
 "encoding/json"
 "os"
 "testing"
 "github.com/charmbracelet/lipgloss"
 "github.com/muesli/termenv"
)
func TestASDGroupDialogReference(t *testing.T){
 lipgloss.SetColorProfile(termenv.TrueColor)
 var frames []map[string]any
 for _,size:=range [][2]int{{120,32},{63,25},{49,25}}{for _,mode:=range []string{"root","subgroup","move"}{for _,field:=range []int{0,1}{
  if mode=="move"&&field==1{continue}
  g:=NewGroupDialog();g.SetSize(size[0],size[1])
  switch mode{case "root":g.ShowCreateWithContextDefaultRoot("project","项目");case "subgroup":g.ShowCreateWithContext("project","项目");case "move":g.ShowMove([]string{"My Sessions","项目","项目/审核"})}
  if mode!="move"&&field==1{g.focusPath()}
  frames=append(frames,map[string]any{"columns":size[0],"rows":size[1],"mode":mode,"field":field,"content":g.View()})
 }}}
 data,err:=json.MarshalIndent(frames,"","  ");if err!=nil{t.Fatal(err)}
 if err=os.WriteFile("/private/tmp/asd-groups-desktop-20261123/upstream-dialog-frames.json",data,0600);err!=nil{t.Fatal(err)}
}
