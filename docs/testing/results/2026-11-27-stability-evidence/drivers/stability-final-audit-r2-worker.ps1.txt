& ([ScriptBlock]::Create([IO.File]::ReadAllText((Join-Path $base 'desktop-api.ps1'))))
. ([ScriptBlock]::Create([IO.File]::ReadAllText((Join-Path $base 'windows-process-metrics.ps1'))))
$roles=@{}
foreach($dir in @(Get-ChildItem $base -Directory -Filter 'stability-ui-product-*-20261127')){
 $role=if($dir.Name-ceq 'stability-ui-product-soak-r1-20261127'){'soak_private'}else{'finished_private'}
 foreach($name in @('asd-instance-r3.exe','asd-stability-fixture.exe','terminal-runtime/psmux.exe','terminal-runtime/OpenConsole.exe')){$file=Join-Path $dir.FullName $name;if(Test-Path $file){$roles[$file]=$role}}
