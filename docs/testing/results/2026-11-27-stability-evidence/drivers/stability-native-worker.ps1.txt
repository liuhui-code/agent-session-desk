$ErrorActionPreference='Stop';$ProgressPreference='SilentlyContinue'
$base=Join-Path $env:USERPROFILE 'asd-acceptance-20261122';$out=Join-Path $base 'stability-native-20261127'
if(Test-Path $out){throw 'Fresh native result directory required'};New-Item -ItemType Directory $out|Out-Null
$m=Get-Content -Raw (Join-Path $base 'windows-tests.json')|ConvertFrom-Json
$results=@()
foreach($item in $m.PSObject.Properties){
 $exe=Join-Path (Join-Path $base 'windows-tests') $item.Name
 if((Get-FileHash $exe).Hash.ToLowerInvariant()-cne $item.Value){throw 'Native test binary changed'}
 $args=@('-test.v','-test.timeout','120s')
 # Helper entrypoints are subprocess-only. Native process tests require separate explicit environment.
 if($item.Name-eq 'instance.test.exe'){$args+=@('-test.run','^Test(ScopeKey|Window|WindowsInstance)')}
 & $exe @args 2>&1|Set-Content (Join-Path $out ($item.Name+'.stdout')) -Encoding utf8
 $exit=$LASTEXITCODE;$lines=@(Get-Content (Join-Path $out ($item.Name+'.stdout')))
 $results+=@{exe=$item.Name;exit_code=$exit;top_pass=@($lines|Where-Object{$_-match '^--- PASS:'}).Count;top_fail=@($lines|Where-Object{$_-match '^--- FAIL:'}).Count;top_skip=@($lines|Where-Object{$_-match '^--- SKIP:'}).Count}
 $results|ConvertTo-Json -Depth 5|Set-Content (Join-Path $out 'progress.json') -Encoding utf8
}
$results|ConvertTo-Json -Depth 5|Set-Content (Join-Path $out 'result.json') -Encoding utf8
