package config_test
import("os";"path/filepath";"testing";"github.com/liuhui-code/agent-session-desk/internal/config")
func TestASDOldUIReaderRejectsV6(t *testing.T){for _,tail:=range []string{"","theme='system'\n"}{p:=filepath.Join(t.TempDir(),"ui.toml");text:="schema_version=6\nsidebar_density='compact'\n"+tail;if e:=os.WriteFile(p,[]byte(text),0600);e!=nil{t.Fatal(e)};if _,e:=config.LoadUI(p);e==nil{t.Fatal("older reader accepted v6")};if e:=config.SaveUI(p,config.UISettings{SidebarDensity:"full"});e==nil{t.Fatal("older writer replaced v6")};b,e:=os.ReadFile(p);if e!=nil||string(b)!=text{t.Fatal("older rejection modified file")}}}
