type asdProfileSourceModel struct{content string}
func (m asdProfileSourceModel) Init()tea.Cmd{return func()tea.Msg{return tea.WindowSizeMsg{Width:120,Height:40}}}
func (m asdProfileSourceModel) Update(tea.Msg)(tea.Model,tea.Cmd){return m,nil}
func (m asdProfileSourceModel) View()string{return m.content}
type asdProfileSourceOutput struct{mu sync.Mutex;data bytes.Buffer;wake chan struct{}}
func (o *asdProfileSourceOutput) Write(p []byte)(int,error){o.mu.Lock();n,e:=o.data.Write(p);o.mu.Unlock();select{case o.wake<-struct{}{}:default:};return n,e}
func (o *asdProfileSourceOutput) snapshot()string{o.mu.Lock();defer o.mu.Unlock();return o.data.String()}
func asdCaptureSource(t *testing.T,content string)string{
 t.Helper();out:=&asdProfileSourceOutput{wake:make(chan struct{},1)}
 p:=tea.NewProgram(asdProfileSourceModel{content},tea.WithInput(nil),tea.WithOutput(out),tea.WithEnvironment([]string{"TERM=xterm-256color"}),tea.WithoutSignalHandler(),tea.WithAltScreen())
 done:=make(chan error,1);go func(){_,e:=p.Run();done<-e}()
 defer func(){
  p.Quit()
  select{case e:= <-done:if e!=nil{t.Errorf("renderer exit: %v",e)};case <-time.After(3*time.Second):p.Kill();<-done;t.Error("renderer did not quit normally")}
 }()
 timer:=time.NewTimer(3*time.Second);defer timer.Stop()
 for{captured:=out.snapshot();if strings.Contains(captured,"ASD_PROFILE_DONE"){return captured};select{case <-out.wake:case <-timer.C:t.Fatalf("source renderer did not complete: %q",captured)}}
}
