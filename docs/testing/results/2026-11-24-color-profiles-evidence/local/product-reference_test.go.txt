type asdProfileProductModel struct{content string}
func (m asdProfileProductModel) Init()tea.Cmd{return nil}
func (m asdProfileProductModel) Update(tea.Msg)(tea.Model,tea.Cmd){return m,nil}
func (m asdProfileProductModel) View()tea.View{v:=tea.NewView(m.content);v.AltScreen=true;return v}
type asdProfileProductOutput struct{mu sync.Mutex;data bytes.Buffer;wake chan struct{}}
func (o *asdProfileProductOutput) Write(p []byte)(int,error){o.mu.Lock();n,e:=o.data.Write(p);o.mu.Unlock();select{case o.wake<-struct{}{}:default:};return n,e}
func (o *asdProfileProductOutput) snapshot()string{o.mu.Lock();defer o.mu.Unlock();return o.data.String()}
func asdCaptureProduct(t *testing.T,content string,profile colorprofile.Profile)string{
 t.Helper();out:=&asdProfileProductOutput{wake:make(chan struct{},1)}
 p:=tea.NewProgram(asdProfileProductModel{content},tea.WithInput(nil),tea.WithOutput(out),tea.WithWindowSize(120,40),tea.WithEnvironment([]string{"TERM=xterm-256color"}),tea.WithoutSignalHandler(),tea.WithColorProfile(profile))
 done:=make(chan error,1);go func(){_,e:=p.Run();done<-e}()
 defer func(){
  p.Quit()
  select{case e:= <-done:if e!=nil{t.Errorf("renderer exit: %v",e)};case <-time.After(3*time.Second):p.Kill();<-done;t.Error("renderer did not quit normally")}
 }()
 timer:=time.NewTimer(3*time.Second);defer timer.Stop()
 for{captured:=out.snapshot();if strings.Contains(captured,"ASD_PROFILE_DONE"){return captured};select{case <-out.wake:case <-timer.C:t.Fatalf("product renderer did not complete: %q",captured)}}
}
