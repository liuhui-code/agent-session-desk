package session
import("encoding/json";"os";"testing")
func TestASDSessionOrderReference(t *testing.T){
 a:=&Instance{ID:"a",Title:"A",GroupPath:"review",Order:0};b:=&Instance{ID:"b",Title:"B",GroupPath:"review",Order:1};c:=&Instance{ID:"c",Title:"C",GroupPath:"other"}
 instances:=[]*Instance{a,b,c};tree:=NewGroupTree(instances);var states []map[string]any
 record:=func(name string,want []string){t.Helper();var ids []string;for _,v:=range tree.Groups["review"].Sessions{ids=append(ids,v.ID)};if len(ids)!=len(want){t.Fatal(ids)};for i:=range want{if ids[i]!=want[i]{t.Fatalf("%s: %v want %v",name,ids,want)}};if tree.Groups["other"].Sessions[0]!=c{t.Fatal("other group changed")};states=append(states,map[string]any{"state":name,"ids":ids,"a_order":a.Order,"b_order":b.Order})}
 record("initial",[]string{"a","b"});tree.MoveSessionUp(a);record("up-boundary",[]string{"a","b"});tree.MoveSessionUp(b);record("swap-up",[]string{"b","a"});tree=NewGroupTree(instances);record("rebuild",[]string{"b","a"});tree.MoveSessionDown(b);record("swap-down",[]string{"a","b"});tree.MoveSessionDown(b);record("down-boundary",[]string{"a","b"})
 data,err:=json.MarshalIndent(states,"","  ");if err!=nil{t.Fatal(err)};if err=os.WriteFile(os.Getenv("ASD_SOURCE_ORDER_RESULT"),data,0600);err!=nil{t.Fatal(err)}
}
